## 공유 인터페이스 (확장)

기존 타입은 유지하고 **shape 종류만 확장**한다.

```js
// Shape: 'bar' | 'prism' | 'L' | 'T' | 'notch'
//   'L','T','notch'는 compound — primitiveVerts에서 직접 정점을 만들지 않고
//   shapes.expandCompound(shape, size) → [{shape:'bar', size, posRel, rotRel}, ...] 로 분해된다.
//
// Occluder.parts: GameStateMachine._occluders가 위 expandCompound 결과를
//   occluder의 (pos, rot)에 합성해 각 part의 absolute pos/rot로 채워 반환.
//   ShadowProjector·Renderer는 단일 part(bar/prism)만 알면 된다 — 변경 0.
//
// Level JSON: movableOccluders[i].shape ∈ 위 5종. compound도 size 한 개로 표현
//   (compound 내부 part 비율은 expandCompound에 하드코딩.)
```

회전축: **z축만** (현재 유지). compound part의 posRel은 occluder의 z축 회전과 동일 각으로 회전시켜 합성한다.

```js
function animateCar(res, done) {
  const traj = res.trajectory;
  let i = 0;
  cancelAnim();
  const step = () => {
    if (i >= traj.length) { done(); return; }
    const [x, y] = traj[i];
    // 다음 샘플과의 차분으로 slope 추정. 마지막 샘플은 직전 slope 유지.
    let slope = 0;
    const j = Math.min(i + 2, traj.length - 1);
    if (j > i) {
      const dx = traj[j][0] - x;
      const dy = traj[j][1] - y;
      if (Math.abs(dx) > 1e-6) slope = Math.atan2(dy, dx);
    }
    renderer.setCar(x, y, slope);
    renderer.render();
    i += 2;
    anim = requestAnimationFrame(step);
  };
  step();
}
```

```js
/**
 * Compound shape를 bar parts 배열로 분해.
 * 각 part는 occluder 로컬 좌표(occluder 중심을 원점으로) 기준 posRel/rotRel을 가진다.
 * occluder 전체에 (pos, rot)이 적용되면 GameStateMachine 쪽에서 posRel을 함께 회전·평행이동.
 * size는 occluder 전체 외접 bounding (sx, sy, sz). 내부 분해 비율은 하드코딩.
 *
 *   L  : ㄴ자 — 아래 가로 bar + 좌측 세로 bar. 두께 t = min(sx,sy) * 0.4
 *   T  : ㅗ자(아래로 줄기) — 위 가로 bar + 중앙 세로 bar. 두께 t = min(sx,sy) * 0.4
 *   notch: U자 — 아래 가로 bar + 좌 세로 bar + 우 세로 bar. 두께 t = min(sx,sy) * 0.35
 *
 * @param {'bar'|'prism'|'L'|'T'|'notch'} shape
 * @param {[number,number,number]} size
 * @returns {Array<{shape:'bar'|'prism', size:[number,number,number], posRel:[number,number,number], rotRel:number}>}
 */
export function expandCompound(shape, size) {
  const [sx, sy, sz] = size;
  if (shape === 'bar' || shape === 'prism') {
    return [{ shape, size: [sx, sy, sz], posRel: [0, 0, 0], rotRel: 0 }];
  }
  if (shape === 'L') {
    const t = Math.min(sx, sy) * 0.4;
    return [
      // 아래 가로 — y는 -sy/2 .. -sy/2 + t (중심 -sy/2 + t/2)
      { shape: 'bar', size: [sx, t, sz], posRel: [0, -sy / 2 + t / 2, 0], rotRel: 0 },
      // 좌측 세로 — x는 -sx/2 .. -sx/2 + t (중심 -sx/2 + t/2), y는 -sy/2..sy/2
      { shape: 'bar', size: [t, sy, sz], posRel: [-sx / 2 + t / 2, 0, 0], rotRel: 0 },
    ];
  }
  if (shape === 'T') {
    const t = Math.min(sx, sy) * 0.4;
    return [
      // 위 가로 — y는 sy/2 - t .. sy/2 (중심 sy/2 - t/2)
      { shape: 'bar', size: [sx, t, sz], posRel: [0, sy / 2 - t / 2, 0], rotRel: 0 },
      // 중앙 세로 — x=0, y 전체. 두께 t.
      { shape: 'bar', size: [t, sy, sz], posRel: [0, 0, 0], rotRel: 0 },
    ];
  }
  if (shape === 'notch') {
    const t = Math.min(sx, sy) * 0.35;
    return [
      // 아래 가로(바닥) — y는 -sy/2 .. -sy/2 + t (중심 -sy/2 + t/2)
      { shape: 'bar', size: [sx, t, sz], posRel: [0, -sy / 2 + t / 2, 0], rotRel: 0 },
      // 좌 기둥 — x는 -sx/2 .. -sx/2 + t, y 전체
      { shape: 'bar', size: [t, sy, sz], posRel: [-sx / 2 + t / 2, 0, 0], rotRel: 0 },
      // 우 기둥 — x는 sx/2 - t .. sx/2, y 전체
      { shape: 'bar', size: [t, sy, sz], posRel: [ sx / 2 - t / 2, 0, 0], rotRel: 0 },
    ];
  }
  throw new Error(`expandCompound: unsupported shape ${shape}`);
}
```

그리고 `_occluders()` 본문을 다음으로 교체:

```js
_occluders() {
  const buildOcc = (src) => {
    // src = { shape, role, size, pos, rot }
    const parts0 = expandCompound(src.shape, src.size);
    const rotDeg = typeof src.rot === 'number' ? src.rot : 0;
    const r = (rotDeg * Math.PI) / 180;
    const cs = Math.cos(r), sn = Math.sin(r);
    const parts = parts0.map((p) => {
      const [pxL, pyL, pzL] = p.posRel;
      // occluder 로컬 (x,y)를 z축 회전 후 occluder 월드 위치에 합성
      const rx = cs * pxL - sn * pyL;
      const ry = sn * pxL + cs * pyL;
      return {
        shape: p.shape,
        size: p.size,
        pos: [src.pos[0] + rx, src.pos[1] + ry, src.pos[2] + pzL],
        rot: rotDeg + p.rotRel,
      };
    });
    return { parts, role: src.role || 'floor' };
  };

  const fixed = this.level.fixedOccluders.map((f) =>
    buildOcc({ shape: f.shape, role: f.role, size: f.size, pos: f.pos, rot: f.rot || 0 })
  );
  const mov = this.movables.map((m) =>
    buildOcc({ shape: m.shape, role: m.role, size: m.size, pos: m.pos, rot: m.rot })
  );
  return fixed.concat(mov);
}
```

> 주의: `m.rot` 가 배열(오일러 [rx,ry,rz] 또는 쿼터니언)인 경우는 z 성분만 사용하거나, compound가 아닐 때만 그대로 통과시킨다. iteration-2는 **z축만**으로 회전축을 한정했으므로 `rot`는 항상 숫자(도). compound이면 더더욱 숫자만. 트랙볼/쿼터니언 입력은 InteractionController 측에서 차단(아래 Task 4.4).

### Task 4.4: 회전축 z축으로 강제 (InteractionController)

**Files:**

> 현재 InteractionController는 트랙볼로 3D 회전(쿼터니언 또는 오일러[rx,ry,rz])을 만들어 `onChange(index, {pos, rot})`로 보낸다. iteration-2 결정에 따라 **z축 회전 도(degree, number)**만 통지하도록 단순화한다. compound shape는 z축 외 회전을 받으면 part posRel 회전이 깨지므로 필수.

- [ ] **Step 1: 트랙볼 누적 상태를 z축 1자유도로**

`src/ui/InteractionController.js` 의 트랙볼 누적 변수(`this.turn` Map, 그리고 `_move` 의 회전 분기)에서 yaw/pitch 두 축 누적을 **z축 회전 단일**로 변경:

```js
// 좌드래그 회전: 마우스 x 이동(px) 누적 → z축 회전(deg) 변화.
//   감도 0.6 deg/px. 회전 방향: 우측 드래그 = +z(반시계).
```

`_move` 의 회전 모드 본문에서, 기존 yaw/pitch 누적 코드 대신 다음 골격으로 교체 (메서드 시그니처는 유지):

```js
// drag.mode === 'rotate' 분기 안에서:
const dxPx = e.clientX - this.drag.lastX;
this.drag.lastX = e.clientX;
const idx = this.drag.index;
const cur = this.turn.get(idx) || { deg: this._initialRotDeg(idx) };
cur.deg = (cur.deg + dxPx * 0.6) % 360;
this.turn.set(idx, cur);
// 메시 즉시 시각 반영
this.drag.mesh.rotation.set(0, 0, (cur.deg * Math.PI) / 180);
this.onChange(idx, {
  pos: [this.drag.mesh.position.x, this.drag.mesh.position.y, this.drag.mesh.position.z],
  rot: cur.deg,    // ★ 숫자(deg)만 통지 — 배열/쿼터니언 미통지
});
```

`_initialRotDeg(idx)` 는 현재 메시의 `rotation.z` (rad)를 deg로 변환해 반환하는 헬퍼(없으면 추가).
